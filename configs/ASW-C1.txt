! ============================================================
! ASW-C1 (Phần của Khánh)
! Khối C – Kết nối: PC-Lab1 (Fa0/1 VLAN40), PC-Lab2 (Fa0/2 VLAN40), LAP-C1 (Fa0/3 VLAN70)
! Uplink: G0/1 → DSW-C1, G0/2 → DSW-C2
! Quản trị VLAN 99: 10.3.99.11/24, GW 10.3.99.1
! ============================================================
hostname ASW-C1
no ip domain-lookup
enable secret <ENABLE_SECRET>
service password-encryption
banner motd #Truy cap trai phep bi cam#
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
interface Vlan1
 no ip address
 shutdown
 exit
!
vtp mode transparent
vlan 40
 name PhongThucHanh
 exit
vlan 70
 name AccessPoint
 exit
vlan 99
 name QuanTri
 exit
vlan 999
 name BlackHole
 exit
!
! --- Uplink lên DSW (G0/1 → DSW-C1, G0/2 → DSW-C2) ---
interface range GigabitEthernet0/1-2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 40,70,99
 no shutdown
 exit
!
! --- PC-Lab1 (Fa0/1) và PC-Lab2 (Fa0/2) – VLAN 40 ---
interface range FastEthernet0/1-2
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown
 exit
!
! --- LAP-C1 (Fa0/3) – VLAN 70 ---
interface FastEthernet0/3
 switchport mode access
 switchport access vlan 70
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
 no shutdown
 exit
!
! --- Cổng không dùng → VLAN 999, shutdown ---
interface range FastEthernet0/4-24
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! --- IP quản trị VLAN 99 ---
interface Vlan99
 ip address 10.3.99.11 255.255.255.0
 no shutdown
 exit
ip default-gateway 10.3.99.1
!
! --- DHCP Snooping ---
ip dhcp snooping
ip dhcp snooping vlan 40,70
no ip dhcp snooping information option
interface range GigabitEthernet0/1-2
 ip dhcp snooping trust
 exit
!
! --- DAI (thử; nếu PT báo lỗi thì bỏ khối này) ---
ip arp inspection vlan 40,70
interface range GigabitEthernet0/1-2
 ip arp inspection trust
 exit
end
write memory
