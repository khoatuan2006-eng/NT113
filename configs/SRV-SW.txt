! ============================================================
! SRV-SW (Phần của Khánh)
! Switch kết nối vùng máy chủ nội bộ
! Uplink: G0/1 → DSW-A1, G0/2 → DSW-A2
! DHCP-DNS (Fa0/1 VLAN50), RADIUS (Fa0/2 VLAN50), PORTAL-NB (Fa0/3 VLAN50)
! WLC1 (Fa0/4): trunk, native VLAN 50, cho phép 30,50
! Quản trị VLAN 99: 10.1.99.13/24, GW 10.1.99.1
! ============================================================
hostname SRV-SW
no ip domain-lookup
enable secret <ENABLE_SECRET>
service password-encryption
banner motd #Truy cap trai phep bi cam#
line console 0
 logging synchronous
 exec-timeout 0 0
 exit
interface Vlan1
 no ip address
 shutdown
 exit
!
vtp mode transparent
vlan 30
 name SinhVien-WiFi
 exit
vlan 50
 name MayChu
 exit
vlan 99
 name QuanTri
 exit
vlan 999
 name BlackHole
 exit
!
! --- Uplink lên DSW-A1 (G0/1) và DSW-A2 (G0/2) ---
interface range GigabitEthernet0/1-2
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,50,99
 no shutdown
 exit
!
! --- DHCP-DNS, RADIUS, PORTAL-NB – VLAN 50 (Access) ---
interface range FastEthernet0/1-3
 switchport mode access
 switchport access vlan 50
 spanning-tree portfast
 no shutdown
 exit
!
! --- WLC1 (Fa0/4): trunk, native VLAN 50, cho phép VLAN 30,50 ---
! WLC management untagged trong VLAN 50 (native 50)
interface FastEthernet0/4
 switchport mode trunk
 switchport trunk native vlan 50
 switchport trunk allowed vlan 30,50
 no shutdown
 exit
!
! --- Cổng không dùng → VLAN 999, shutdown ---
interface range FastEthernet0/5-24
 switchport mode access
 switchport access vlan 999
 shutdown
 exit
!
! --- IP quản trị VLAN 99 ---
interface Vlan99
 ip address 10.1.99.13 255.255.255.0
 no shutdown
 exit
ip default-gateway 10.1.99.1
end
write memory
